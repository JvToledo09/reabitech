    <script>
    (function() {
        const INTERVALO = 30000; // 30s
        const URL_API = '{% url "dashboard:api_notificacoes_count" %}';

        const URL_MARCAR_LIDA_BASE = '/dashboard/notificacoes/';
        const URL_MARCAR_LIDA_SUFIXO = '/lida/';

        function urlMarcarLida(id) {
            return URL_MARCAR_LIDA_BASE + id + URL_MARCAR_LIDA_SUFIXO;
        }

        async function atualizarNotificacoes() {
            try {
                const resp = await fetch(URL_API, {
                    headers: { 'X-Requested-With': 'XMLHttpRequest' },
                    credentials: 'same-origin',
                });
                if (!resp.ok) return;
                const data = await resp.json();

                const sino = document.querySelector('.dropdown .btn-outline-custom .fa-bell')?.closest('button');
                if (sino) {
                    let badge = sino.querySelector('.badge.bg-danger');
                    if (data.total > 0) {
                        if (!badge) {
                            badge = document.createElement('span');
                            badge.className = 'position-absolute top-0 start-100 translate-middle badge rounded-pill bg-danger';
                            badge.style.fontSize = '10px';
                            sino.appendChild(badge);
                        }
                        badge.textContent = data.total;
                    } else if (badge) {
                        badge.remove();
                    }
                }

                const lista = document.querySelector('ul.dropdown-menu[style*="min-width: 320px"]');
                if (lista && data.ultimas.length > 0) {
                    const header = lista.querySelector('li:first-child');
                    const footer = lista.querySelector('li:last-child');
                    const itensAntigos = lista.querySelectorAll('li:not(:first-child):not(:last-child)');
                    itensAntigos.forEach(i => i.remove());

                    data.ultimas.forEach(n => {
                        const li = document.createElement('li');
                        li.innerHTML = `
                            <a class="dropdown-item py-3 border-bottom" href="${urlMarcarLida(n.id)}">
                                <div class="d-flex align-items-start">
                                    <div class="bg-primary bg-opacity-10 rounded-circle p-2 me-2" style="width: 35px; height: 35px; display: flex; align-items: center; justify-content: center;">
                                        <i class="fas fa-info-circle text-primary small"></i>
 </div>
                                    <div>
                                        <strong class="small d-block">${n.titulo}</strong>
                                        <small class="text-muted d-block">${n.mensagem.substring(0, 60)}</small>
                                        <small class="text-muted" style="font-size: 11px;">${n.criada_em}</small>
 </div>
 </div>
 </a>
                        `;
                        if (footer) {
                            lista.insertBefore(li, footer);
                        } else {
                            lista.appendChild(li);
                        }
                    });
                }
            } catch (e) {
                console.debug('Polling notificações falhou:', e);
            }
        }

        setTimeout(atualizarNotificacoes, 2000);
        setInterval(atualizarNotificacoes, INTERVALO);
    })();
 </script>

    //  Função auxiliar para ler dados do json_script com segurança
    function getJSON(id) {
        const element = document.getElementById(id);
        return element ? JSON.parse(element.textContent) : [];
    }

    // ==========================================
    // GRÁFICO 1: Evolução da Recuperação
    // ==========================================
    const evoLabels = getJSON('evo_labels');
    const evoData = getJSON('evo_data');

    if (evoLabels.length > 0) {
        new Chart(document.getElementById('evolutionChart').getContext('2d'), {
            type: 'line',
            data: {
                labels: evoLabels,
                datasets: [{
                    label: '% Recuperação',
                    data: evoData,
                    borderColor: '#2BA181',
                    backgroundColor: 'rgba(43, 161, 129, 0.1)',
                    fill: true,
                    tension: 0.4
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: { legend: { display: false } }
            }
        });
    } else {
        showEmptyChart('evolutionChart', 'Sem evoluções registradas ainda');
    }

    // ==========================================
    // GRÁFICO 2: Status dos Atletas
    // ==========================================
    const statusData = getJSON('status_data');
    const hasStatusData = statusData.some(v => v > 0);

    if (hasStatusData) {
        new Chart(document.getElementById('statusChart').getContext('2d'), {
            type: 'pie',
            data: {
                labels: getJSON('status_labels'),
                datasets: [{
                    data: statusData,
                    backgroundColor: ['#dc3545', '#2BA181', '#ffc107'],
                    borderWidth: 0
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: { legend: { position: 'bottom' } }
            }
        });
    } else {
        showEmptyChart('statusChart', 'Sem dados de status ainda');
    }

    // ==========================================
    // GRÁFICO 3: Lesões por Modalidade
    // ==========================================
    const modalData = getJSON('modal_data');
    if (modalData.some(v => v > 0)) {
        new Chart(document.getElementById('modalChart').getContext('2d'), {

            data: {
                labels: getJSON('modal_labels'),
                datasets: [{
                    data: modalData,

                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: { legend: { display: false } },

window.showToast = function(title, message, type = 'success') {
    const container = document.getElementById('toastContainer');
    if (!container) return;

    const icons = {
        success: 'fa-check',
        error: 'fa-times',
        warning: 'fa-exclamation',
        info: 'fa-info'
    };

    const toast = document.createElement('div');
    toast.className = `toast-item ${type}`;
    toast.innerHTML = `
        <div class="toast-icon">
            <i class="fas ${icons[type] || icons.info}"></i>
 </div>
        <div class="toast-content">
            <div class="toast-title">${title}</div>
            <p class="toast-message">${message}</p>
 </div>
    `;

    container.appendChild(toast);

    setTimeout(() => {
        toast.classList.add('hiding');
        setTimeout(() => toast.remove(), 300);
    }, 4000);
};
